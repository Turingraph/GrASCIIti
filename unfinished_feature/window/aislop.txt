#include "window.h"

/*
 * time : O(1)
 * space: O(1)
 */
void	update_camera(mlx_key_data_t keydata, t_2d_camera *camera)
{
	if (camera == NULL || is_valid_key(keydata) == false)
		return ;
	if (keydata.key == MLX_KEY_UP)
		camera->offset.y -= 5;
	else if (keydata.key == MLX_KEY_DOWN)
		camera->offset.y += 5;
	else if (keydata.key == MLX_KEY_LEFT)
		camera->offset.x -= 5;
	else if (keydata.key == MLX_KEY_RIGHT)
		camera->offset.x += 5;
	else if (keydata.key == MLX_KEY_1)
	{
		camera->zoom *= 1.01f;
		if (camera->zoom > 3.0f)
			camera->zoom = 3.0f;
	}
	else if (keydata.key == MLX_KEY_2)
	{
		camera->zoom /= 1.01f;
		if (camera->zoom < 0.2f)
			camera->zoom = 0.2f;
	}
}

/*
 * Draw every visible horizontal and vertical grid line exactly once.
 *
 * time : O(n)
 * space: O(1)
 */
void	picture_at_an_exhibition(t_2d_hook *hook)
{
	t_line	interier_tile;
	t_line	line;
	t_area	area;
	int		ix;
	int		iy;

	if (is_2dhook_valid(hook) == false)
		return ;

	interier_tile.p1 = get_first_interier_tile(
		hook->tiles,
		*(hook->camera)
	);
	interier_tile.p2 = get_last_interier_tile(
		hook->tiles,
		*(hook->camera),
		hook->img->width,
		hook->img->height
	);

	if (interier_tile.p1.x < 0
		|| interier_tile.p1.y < 0
		|| interier_tile.p2.x < 0
		|| interier_tile.p2.y < 0)
		return ;

	area = get_all_area(
		hook->img->height,
		hook->img->width
	).sub_area;

	/*
	 * Clear the image once.
	 *
	 * Replace this with your own fast image-clear function if
	 * color_background_mlx() is not appropriate here.
	 */
	color_background_mlx(hook->img, hook->background);

	iy = interier_tile.p1.y;
	while (iy <= interier_tile.p2.y)
	{
		line.p1 = get_ith_tile_screen(
			hook->tiles,
			*(hook->camera),
			interier_tile.p1.x,
			iy
		);
		line.p2 = get_ith_tile_screen(
			hook->tiles,
			*(hook->camera),
			interier_tile.p2.x,
			iy
		);
		draw_mlx_straight_line(
			hook->img,
			line,
			area,
			hook->ink
		);
		iy++;
	}

	ix = interier_tile.p1.x;
	while (ix <= interier_tile.p2.x)
	{
		line.p1 = get_ith_tile_screen(
			hook->tiles,
			*(hook->camera),
			ix,
			interier_tile.p1.y
		);
		line.p2 = get_ith_tile_screen(
			hook->tiles,
			*(hook->camera),
			ix,
			interier_tile.p2.y
		);
		draw_mlx_straight_line(
			hook->img,
			line,
			area,
			hook->ink
		);
		ix++;
	}
}

/*
 * time : O(n)
 * space: O(1)
 */
void	hook_pan_and_zoom(mlx_key_data_t keydata, void *param)
{
	t_2d_hook	*hook;

	hook = (t_2d_hook *)param;
	if (is_2dhook_valid(hook) == false
		|| is_valid_key(keydata) == false)
		return ;

	if (keydata.key == MLX_KEY_ESCAPE)
	{
		mlx_close_window(hook->mlx);
		return ;
	}

	update_camera(keydata, hook->camera);
	picture_at_an_exhibition(hook);
}
