#include"paint.h"

// time : O(n)
// space: O(1)
void	fill_cells_height(
	t_table_fdf *dst,
	int height,
	bool is_overwrite,
	bool(*is_filtered_cell)(const t_table_fdf *dst, size_t index))
{
	size_t	i;

	i = 0;
	while (dst != NULL && i < dst->row * dst->col && dst->arr != NULL)
	{
		if ((is_filtered_cell == NULL || is_filtered_cell(dst, i) == true)
			&& ((is_overwrite == false && dst->arr[i] <= 0) || is_overwrite == true))
			dst->arr[i] = height;
		i += 1;
	}
}

// time : O(n)
// space: O(1)
void	fill_cells_color(
	t_table_fdf *dst,
	unsigned char input_value,
	t_enum_rgba rgba_type,
	bool(*is_filtered_cell)(const t_table_fdf *dst, size_t index))
{
	size_t			i;
	unsigned char	*arr;

	arr = get_rgba_of_table_fdf(dst, rgba_type);
	i = 0;
	while (dst != NULL && i < dst->row * dst->col && arr != NULL)
	{
		if (is_filtered_cell == NULL || is_filtered_cell(dst, i) == true)
			arr[i] = input_value;
		i += 1;
	}
}

// time : O(n)
// space: O(1)
void	generate_cells_color(
	t_table_fdf *dst,
	t_enum_rgba channel,
	bool(*is_filtered_cell)(const t_table_fdf *dst, size_t index),
	int(*gen_color)(const t_table_fdf *dst, size_t index))
{
	size_t			i;
	int				height;
	unsigned char	*arr;

	arr = get_rgba_of_table_fdf(dst, channel);
	i = 0;
	while (dst != NULL && i < dst->row * dst->col)
	{
		if (is_filtered_cell == NULL || is_filtered_cell(dst, i) == true)
		{
			height = 0;
			if (gen_color != NULL)
				height = gen_color(dst, i);
			if (channel == HEIGHT && dst->arr != NULL)
				dst->arr[i] = height;
			else if (arr != NULL)
				arr[i] = (unsigned char)(height % 256);
		}
		i += 1;
	}
}

// time : O(1)
// space: O(1)
bool	is_empty_space(const t_table_fdf *src, size_t index)
{
	if (src == NULL || src->arr == NULL || index >= src->row * src->col)
		return (false);
	if (src->arr[index] <= 0)
		return (true);
	return (false);
}

// time : O(n)
// space: O(1)
void	paint_table_background(t_table_fdf *dst, t_rgba color)
{
	fill_cells_color(dst, color.r, RED, is_empty_space);
	fill_cells_color(dst, color.g, GREEN, is_empty_space);
	fill_cells_color(dst, color.b, BLUE, is_empty_space);
	fill_cells_color(dst, color.a, ALPHA, is_empty_space);
}
