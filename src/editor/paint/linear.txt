// time : O(1)
// space: O(1)
size_t	length_linear_gradient_fdf(t_gradient *src, float shift, float scale)
{
	return ((size_t)(f_floor(src->x2 - src->x1)/scale) + (size_t)f_floor(1 - shift));
}

// time : O(n)
// space: O(n)
t_gradient	**create_linear_gradient_fdf(float scale, float shift,
	float percent, t_gradient *src)
{
	size_t		len;
	t_gradient	**dst;
	size_t		i;

	if (scale == 0)
		return (NULL);
	len = length_linear_gradient_fdf(src, shift, scale);
	dst = (t_gradient **)malloc_talk(sizeof(t_gradient *) * len,
		"paint/linear.c/create_linear_gradient\n");
	if (dst == NULL)
		return (NULL);
	i = 0;
	while (i < len)
	{
		dst[i] = (t_gradient *)malloc_talk(sizeof(t_gradient),
			"paint/linear.c/create_linear_gradient\n");
		if (dst[i] == NULL)
		{
			free_nest_arr((void **)dst, i);
			return (NULL);
		}
		f_gradient_cpy(src, dst[i]);
		dst[i]->x1 = (int)f_floor(scale * (i - percent + shift));
		dst[i]->x2 = (int)f_floor(scale * (i + shift));
		i += 1;
	}
	return (dst);
}

// time : O(len * row * col)
// space: O(1)
void	paint_multi_gradients_fdf(t_table_fdf *table, size_t len, t_gradient **rgb, char dim)
{
	size_t	i;

	i = 0;
	while (i < len)
	{
		paint_gradient_fdf(table, rgb[i], dim);
		i += 1;
	}
}
