// // time : O(1)
// // space: O(1)
size_t	get_window_dimension_fdf(size_t row, size_t col, char axis)
{
	size_t	window_size;

	window_size = 1080;
	if (row > window_size || col > window_size || row * col == 0)
		return (0);
	if (row == col || (row < col && axis == 0) || (row > col && axis == 1))
		return (window_size);
	if (row > col && axis == 0)
		return (col * (size_t)f_floor((float)window_size / (float)row));
	if (row < col && axis == 1)
		return (row * (size_t)f_floor((float)window_size / (float)col));
	return (0);
}

// time : O(n)
// space: O(1)
mlx_image_t	*solve_1st_mlx_error(mlx_t *mlx, const t_table_fdf *src)
{
	mlx_image_t	*dst;
	size_t		window_size;

	window_size = 1080;
	if (mlx == NULL || src->row * src->col == 0
		|| src->row > window_size || src->col > window_size)
	{
		if (mlx != NULL)
			mlx_terminate(mlx);
		return (NULL);
	}
	if (src->row >= src->col)
		dst = mlx_new_image(mlx, (window_size / src->row) * src->col,
			window_size);
	else
		dst = mlx_new_image(mlx, window_size,
			(window_size / src->col) * src->row);
	write(1, "dst->width = ", 14);
	ft_putnbr_fd(dst->width, 1, "0123456789", 1);
	write(1, "\n", 1);
	write(1, "src->col = ", 12);
	ft_putnbr_fd(src->col, 1, "0123456789", 1);
	write(1, "\n", 1);
	if (dst == NULL)
	{
		mlx_terminate(mlx);
		return(NULL);
	}
	return (dst);
}

// time : O(n)
// space: O(n)
int	view_table_fdf(const t_table_fdf *src)
{
	mlx_t		*mlx;
	mlx_image_t	*img;

	mlx = mlx_init(1920, 1080, "Subset at 4:42pm", true);
	img = solve_1st_mlx_error(mlx, src);
	if (img == NULL)
		return (-1);
	paint_table_on_image(src, img);
	if (-1 == mlx_image_to_window(mlx, img,
		(1920 - img->width) / 2, (1080 - img->height) / 2))
	{
		mlx_terminate(mlx);
		return(-1);
	}
	mlx_loop(mlx);
	mlx_delete_image(mlx, img);
	mlx_terminate(mlx);
	return (1);
}
